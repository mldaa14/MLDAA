# Non-Recursive Fibonacci Series

nterms = int(input("How many terms? "))

n1, n2 = 0, 1
count = 0

if nterms <= 0:
    print("Please enter a positive integer")

else:
    print("Fibonacci sequence:")

    while count < nterms:
        print(n1, end=" ")

        nth = n1 + n2
        n1 = n2
        n2 = nth

        count += 1




# Recursive Fibonacci Series

def recur_fibo(n):
    if n <= 1:
        return n
    else:
        return recur_fibo(n - 1) + recur_fibo(n - 2)


nterms = int(input("How many terms? "))

if nterms <= 0:
    print("Please enter a positive integer")

else:
    print("Fibonacci sequence:")

    for i in range(nterms):
        print(recur_fibo(i), end=" ")

# Fibonacci Series using Recursive and Non-Recursive Methods

# Non-Recursive Fibonacci Function
def non_recursive_fibo(n):
    n1, n2 = 0, 1
    count = 0

    print("Non-Recursive Fibonacci Series:")

    while count < n:
        print(n1, end=" ")

        nth = n1 + n2
        n1 = n2
        n2 = nth

        count += 1

    print()


# Recursive Fibonacci Function
def recur_fibo(n):
    if n <= 1:
        return n
    else:
        return recur_fibo(n - 1) + recur_fibo(n - 2)


# Main Program
nterms = int(input("How many terms? "))

if nterms <= 0:
    print("Please enter a positive integer")

else:
    non_recursive_fibo(nterms)

    print("Recursive Fibonacci Series:")
    for i in range(nterms):
        print(recur_fibo(i), end=" ")
    print()
